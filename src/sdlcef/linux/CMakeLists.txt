cmake_minimum_required(VERSION 3.24)
project(qrazy_sdl_cef_gate LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CEF_ROOT "" CACHE PATH "Pinned CEF minimal distribution")
set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
find_package(SDL3 CONFIG REQUIRED)
list(REMOVE_ITEM CEF_STANDARD_LIBS X11) # Wayland host has no Xlib calls; CEF retains its own runtime dependencies.
list(FILTER CEF_LINKER_FLAGS EXCLUDE REGEX "^-Wl,-rpath,[.]$")
add_subdirectory(${CEF_LIBCEF_DLL_WRAPPER_PATH} libcef_dll_wrapper)
add_library(libcef_lib SHARED IMPORTED)
set_target_properties(libcef_lib PROPERTIES IMPORTED_LOCATION "${CEF_ROOT}/Release/libcef.so")
add_executable(qrazy-sdl-cef app.cpp)
SET_EXECUTABLE_TARGET_PROPERTIES(qrazy-sdl-cef)
target_include_directories(qrazy-sdl-cef PRIVATE "${CEF_ROOT}")
target_link_libraries(qrazy-sdl-cef libcef_lib libcef_dll_wrapper SDL3::SDL3 EGL GLESv2 ${CEF_STANDARD_LIBS})
set_target_properties(qrazy-sdl-cef PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
