cmake_minimum_required(VERSION 3.24)
project(qrazy_windows_port LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CEF_ROOT "" CACHE PATH "Pinned Windows CEF distribution")
set(SDL_ROOT "" CACHE PATH "Pinned SDL VC development distribution")
set(USE_SANDBOX ON CACHE BOOL "Required Windows sandbox" FORCE)
set(CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory(${CEF_LIBCEF_DLL_WRAPPER_PATH} libcef_dll_wrapper)
ADD_LOGICAL_TARGET(libcef_lib "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
set(QRAZY_ICON_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../../assets/qrazy.ico")
configure_file(qrazy.rc "${CMAKE_CURRENT_BINARY_DIR}/qrazy.rc" @ONLY)
add_library(Qrazy SHARED app_windows.cpp "${CMAKE_CURRENT_BINARY_DIR}/qrazy.rc")
SET_LIBRARY_TARGET_PROPERTIES(Qrazy)
target_include_directories(Qrazy PRIVATE "${CEF_ROOT}" "${SDL_ROOT}/include")
target_link_libraries(Qrazy libcef_lib libcef_dll_wrapper "${SDL_ROOT}/lib/x64/SDL3.lib" d3d11.lib dxgi.lib advapi32.lib iphlpapi.lib ws2_32.lib ${CEF_STANDARD_LIBS})
target_compile_options(Qrazy PRIVATE /W4 /WX)
target_compile_definitions(Qrazy PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(QrazyLauncherObject OBJECT launcher.cpp)
target_compile_options(QrazyLauncherObject PRIVATE /W4 /WX /EHsc)
add_executable(QrazyPortable WIN32 $<TARGET_OBJECTS:QrazyLauncherObject> "${CMAKE_CURRENT_BINARY_DIR}/qrazy.rc")
set_target_properties(QrazyPortable PROPERTIES OUTPUT_NAME Qrazy)
target_link_libraries(QrazyPortable PRIVATE bcrypt.lib user32.lib advapi32.lib)

set(QRAZY_CONTROL_DIR "" CACHE PATH "Prepared public updater controls and generated pins")
target_include_directories(Qrazy PRIVATE "${QRAZY_CONTROL_DIR}")
target_include_directories(QrazyLauncherObject PRIVATE "${QRAZY_CONTROL_DIR}")

# Reuse the existing source-selected destination; no command-line runtime override.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../client-mode.cjs" QRAZY_CLIENT_MODE_SOURCE)
if(QRAZY_CLIENT_MODE_SOURCE MATCHES "const CLIENT_MODE = 'dev';")
  set(QRAZY_DEV_MODE 1)
elseif(QRAZY_CLIENT_MODE_SOURCE MATCHES "const CLIENT_MODE = 'live';")
  set(QRAZY_DEV_MODE 0)
else()
  message(FATAL_ERROR "Invalid client-mode.cjs destination")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../client-mode.cjs")
# Dev-only "true client": a Vulkan renderer (viewer.cpp, kept outside the repository) drawn into the window, toggled with F2. Off unless QRAZY_TRUE_CLIENT_DIR is set, and refused for live builds.
set(QRAZY_TRUE_CLIENT_DIR "" CACHE PATH "Dev-only true-client renderer sources (empty = off)")
set(QRAZY_CORE_DIR "" CACHE PATH "Render core directory (render_core.h, out/render_core.lib)")
set(QRAZY_CORE_LIB "" CACHE FILEPATH "render_core.lib (default: QRAZY_CORE_DIR/out/render_core.lib)")
if(NOT QRAZY_CORE_LIB)
  set(QRAZY_CORE_LIB "${QRAZY_CORE_DIR}/out/render_core.lib")
endif()
set(QRAZY_VK_INCLUDE "" CACHE PATH "Vulkan headers")
if(QRAZY_TRUE_CLIENT_DIR)
  if(NOT QRAZY_DEV_MODE)
    message(FATAL_ERROR "QRAZY_TRUE_CLIENT_DIR needs CLIENT_MODE dev")
  endif()
  target_sources(Qrazy PRIVATE "${QRAZY_TRUE_CLIENT_DIR}/viewer.cpp")
  set_source_files_properties("${QRAZY_TRUE_CLIENT_DIR}/viewer.cpp" PROPERTIES COMPILE_OPTIONS "/W3;/WX-;/wd4996")
  target_compile_definitions(Qrazy PRIVATE QRAZY_TRUE_CLIENT=1)
  target_include_directories(Qrazy PRIVATE "${QRAZY_CORE_DIR}" "${QRAZY_CORE_DIR}/text" "${QRAZY_VK_INCLUDE}")
  target_link_libraries(Qrazy "${QRAZY_CORE_LIB}" "${QRAZY_CORE_DIR}/text/out/native/text_engine.lib" winmm.lib ole32.lib windowscodecs.lib uuid.lib gdi32.lib user32.lib)
endif()

target_compile_definitions(Qrazy PRIVATE QRAZY_DEV_MODE=${QRAZY_DEV_MODE})
target_compile_definitions(QrazyLauncherObject PRIVATE QRAZY_DEV_MODE=${QRAZY_DEV_MODE})
