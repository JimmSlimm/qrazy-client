cmake_minimum_required(VERSION 3.24)
project(qrazy_windows_port LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CEF_ROOT "" CACHE PATH "Pinned Windows CEF distribution")
set(SDL_ROOT "" CACHE PATH "Pinned SDL VC development distribution")
set(USE_SANDBOX ON CACHE BOOL "Required Windows sandbox" FORCE)
set(CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory(${CEF_LIBCEF_DLL_WRAPPER_PATH} libcef_dll_wrapper)
ADD_LOGICAL_TARGET(libcef_lib "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
set(QRAZY_ICON_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../../assets/qrazy.ico")
configure_file(qrazy.rc "${CMAKE_CURRENT_BINARY_DIR}/qrazy.rc" @ONLY)
add_library(Qrazy SHARED app_windows.cpp "${CMAKE_CURRENT_BINARY_DIR}/qrazy.rc")
SET_LIBRARY_TARGET_PROPERTIES(Qrazy)
target_include_directories(Qrazy PRIVATE "${CEF_ROOT}" "${SDL_ROOT}/include")
target_link_libraries(Qrazy libcef_lib libcef_dll_wrapper "${SDL_ROOT}/lib/x64/SDL3.lib" d3d11.lib dxgi.lib ${CEF_STANDARD_LIBS})
target_compile_options(Qrazy PRIVATE /W4 /WX)
target_compile_definitions(Qrazy PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(QrazyLauncherObject OBJECT launcher.cpp)
target_compile_options(QrazyLauncherObject PRIVATE /W4 /WX /EHsc)
add_executable(QrazyPortable WIN32 $<TARGET_OBJECTS:QrazyLauncherObject> "${CMAKE_CURRENT_BINARY_DIR}/qrazy.rc")
set_target_properties(QrazyPortable PROPERTIES OUTPUT_NAME Qrazy)
target_link_libraries(QrazyPortable PRIVATE bcrypt.lib user32.lib advapi32.lib)

set(QRAZY_CONTROL_DIR "" CACHE PATH "Prepared public updater controls and generated pins")
target_include_directories(Qrazy PRIVATE "${QRAZY_CONTROL_DIR}")
target_include_directories(QrazyLauncherObject PRIVATE "${QRAZY_CONTROL_DIR}")
